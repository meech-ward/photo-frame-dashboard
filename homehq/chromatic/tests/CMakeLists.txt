cmake_minimum_required(VERSION 3.22)
project(homehq_native_checks C)
if(NOT MCU_ROOT OR NOT IDF_ROOT)
  message(FATAL_ERROR "Pass -DMCU_ROOT=<prepared isolated MCU> -DIDF_ROOT=<ESP-IDF v5.3>")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
add_subdirectory(${MCU_ROOT}/managed_components/lvgl__lvgl lvgl EXCLUDE_FROM_ALL)
add_library(cjson STATIC ${IDF_ROOT}/components/json/cJSON/cJSON.c)
target_include_directories(cjson PUBLIC ${IDF_ROOT}/components/json/cJSON)
add_executable(pack_check test_pack.c ../src/homehq_pack.c)
target_include_directories(pack_check PRIVATE ../src)
target_compile_options(pack_check PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME cache_roundtrip COMMAND pack_check ${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures/demo.json)
add_executable(realtime_check test_realtime.c ../src/homehq_realtime.c)
target_include_directories(realtime_check PRIVATE realtime_stubs ../src ${MCU_ROOT}/managed_components/lvgl__lvgl)
target_link_libraries(realtime_check PRIVATE lvgl cjson)
add_test(NAME private_realtime COMMAND realtime_check)
add_executable(service_check test_service.c ../src/homehq_pack.c)
target_include_directories(service_check PRIVATE ../src . ${MCU_ROOT}/managed_components/lvgl__lvgl)
target_link_libraries(service_check PRIVATE lvgl cjson)
target_compile_options(service_check PRIVATE -Wall -Wextra -Werror)
add_test(NAME auth_outbox COMMAND service_check)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test_ui.c")
  add_executable(ui_check test_ui.c ../src/homehq_ui.c ${MCU_ROOT}/main/homehq_demo.c ${MCU_ROOT}/main/homehq_assets.c)
  target_include_directories(ui_check PRIVATE ../src ${MCU_ROOT}/main ${MCU_ROOT}/tests/host/stubs ${MCU_ROOT}/components/button)
  target_link_libraries(ui_check PRIVATE lvgl cjson m)
  add_test(NAME navigation COMMAND ui_check ${CMAKE_CURRENT_BINARY_DIR})
endif()
